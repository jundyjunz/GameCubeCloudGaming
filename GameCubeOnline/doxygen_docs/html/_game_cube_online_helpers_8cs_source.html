26 //This class exists because a websocket in C# needs to recieve messages in order to see that the state of a websocket is closed.
27 // therefore we have this watchdog class for websockets that only send information.
28 private class WebSocketWatchDog : IDisposable {
29 protected CancellationTokenSource myCancellationTokenSource;
30 protected WebSocket mySocket;
31 protected int myRecieveMessageSize;
32 public CancellationTokenSource CTS { get => myCancellationTokenSource; }
33
34 public WebSocketWatchDog(WebSocket aSocket, int theRecieveMessageSize=4) {
35 mySocket=aSocket;
36 myCancellationTokenSource=new CancellationTokenSource();
37 myRecieveMessageSize = theRecieveMessageSize;
38 Task.Run(async () => await detectWebsocketClose());
39
40 }

42 protected async Task detectWebsocketClose()
43 {
44 try {
45 // ArrayPool is basically a memory pool you can use instead of making allocations at every turn.
46 // Its better than just allocating a bunch of bytes.
47 byte[] theTempArray = ArrayPool<byte>.Shared.Rent(myRecieveMessageSize);
48 WebSocketReceiveResult theMessage;
49 do { theMessage = await mySocket.ReceiveAsync(theTempArray, myCancellationTokenSource.Token); } // these guys need a cancellation token cause they can poll indefinitely. The cancellation token will break them out of their polling loop.
50 while (mySocket.State == WebSocketState.Open && theMessage.MessageType != WebSocketMessageType.Close);
51 ArrayPool<byte>.Shared.Return(theTempArray);
52 }
53 catch { }
54 finally { myCancellationTokenSource.Cancel(); }
55 }
56
57 public void Dispose() { myCancellationTokenSource.Dispose(); }

66 public async static Task sendFrame<T> (IServiceProvider aService, int aClientId, HttpContext aContext) where T: Capture<T>
67 {
68 WebSocket theWebSocket = await aContext.WebSockets.AcceptWebSocketAsync();
69 int theFrameRate = (typeof(T)==typeof(Capture<CaptureVideo>)) ? aService.GetRequiredService<GameCubeOnlineSettings>().ClientVideoFrameRate : aService.GetRequiredService<GameCubeOnlineSettings>().ClientAudioFrameRate;
70 Capture<T> theCapture = aService.GetRequiredService<T>();
71
72 WebSocketWatchDog theWatchDog = new WebSocketWatchDog(theWebSocket);
73
74 try
75 {
76 while (theWebSocket.State == WebSocketState.Open && !theWatchDog.CTS.Token.IsCancellationRequested)
77 {
78 ReadOnlyMemory<byte>? theFrameBytes = theCapture.readFromBuffer(aClientId);
79 if (theFrameBytes.HasValue) await theWebSocket.SendAsync(theFrameBytes.Value, WebSocketMessageType.Binary, true, theWatchDog.CTS.Token);
80 await Task.Delay(theFrameRate);
81 }
82 }
83 catch (Exception aException) { Console.WriteLine($"{typeof(T).Name} {aClientId} Encountered Error: {aException.Message}"); }
84 finally {
85 await killSocket(theWebSocket, typeof(T).Name, aClientId);
86 theCapture.unsubscribeFromBuffer(aClientId);
87 }
88 }

102 Memory<byte> theCommandBufferMemory = new Memory<byte>(theCommandBuffer, 0, theBytesToWrite);
103 var theMessage=await theWebSocket.ReceiveAsync(theCommandBufferMemory, theSocketTimeout.Token);
104 theSerialConnection.readCommand(theCommandBufferMemory, aClientId );
105 if (theMessage.MessageType == WebSocketMessageType.Close) break;
