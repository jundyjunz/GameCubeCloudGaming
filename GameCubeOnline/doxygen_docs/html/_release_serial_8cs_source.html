31 protected byte crc8(ReadOnlySpan<byte> theBytes)
32 {
33 byte theEvolvingCRC = 0x00;
34 for (int i = 0; i < theBytes.Length; i++)
35 {
36 byte theCurrentByte = theBytes[i];
37 for (int j = 0; j < 8; j++)
38 {
39 int theAreBothBitsDifferent = (theEvolvingCRC ^ theCurrentByte) & 0x01;
40 theEvolvingCRC >>= 1;
41 if (theAreBothBitsDifferent==1) theEvolvingCRC ^= myPolynomialMultiplier;
42 theCurrentByte >>= 1;
43 }
44 }
45 return theEvolvingCRC;
46 }

57 // needed such that the enumeration while looping through the subscribers to coalesce the command doesnt fail.

71 public void readCommand(ReadOnlyMemory<byte> aBytes, int aClientId) { aBytes.CopyTo(myCurrentCommands[aClientId]); } // not adding lock here, doesnt seem to cause any issues when client is playing, and lock adds overhead.
72 protected void writeCommand()
73 {
74 // Write command can act faster than read command and may be scheduled more often.
75 // Resetting each client's command in myCurrentCommands will in this case cause stutter
76 // this is because after one iteration acts and resets, the command will coalesce to 0s from the previous reset.
77 lock (mySubscriberLock) foreach (var aCommand in myCurrentCommands) byteArrayOrEquals(myCoalescedCommand, aCommand.Value); //coalescing commands
78 myCoalescedCommand[myBytesToWrite - 1] = crc8(myCoalescedCommand.AsSpan(0, myBytesToWrite-1));
79 myPort.Write(myCoalescedCommand, 0, myBytesToWrite);
80 resetByteArrayExceptFirstAndLastByte(myCoalescedCommand);// have to reset command every time so stale command doesnt persist
81 }
82 public void write(){ while (true) { writeCommand(); Thread.Sleep(myCommandSendSleepTime); } } // minimum thread stall so as not to hog the CPU.

121 {
122 Console.WriteLine($"Attempting To Connect To Port {aPortName} ");
123 SerialPort thePort = new SerialPort(aPortName, myBaudRate);
124 try
125 {
126 thePort.ReadTimeout = myReadTimeout;
127 thePort.WriteTimeout = myWriteTimeout;
128 byte[] theMessageToWrite = [aConnectCode];
129 byte[] theMessageToRead = new byte[1];
130 thePort.DtrEnable = true;// resets the arduino to pre flashed state.
131 thePort.Open();
132 Thread.Sleep(aSleepBuffer);
133 thePort.DiscardInBuffer();
134 thePort.Write(theMessageToWrite, 0, 1);
135 Thread.Sleep(aWaitBuffer); // need to wait a bit here so we can wait for the I/O request.
136 thePort.Read(theMessageToRead, 0, 1);
137 if (theMessageToRead[0] == theMessageToWrite[0]) return registerPort(thePort, aByteCommandLen, myCommandSendSleepTime);
138 thePort.Close();
139 thePort.Dispose();
140 return false;
141
142 }
143 catch (Exception theException)

145 Console.WriteLine(theException.Message);
146 if (thePort.IsOpen) thePort.Close();
147 thePort.Dispose();
148 return false;
